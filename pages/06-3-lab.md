       ├─► [18.5 ~ 22.9: 정상] ─────► "현재 건강한 표준 체중을 훌륭하게 유지하고 계십니다."
       ├─► [23.0 ~ 24.9: 비만 전단계] ─► "하루 30분 유산소 운동과 야식 줄이기를 권장합니다."
       └─► [25.0 이상: 비만] ───────► "심혈관 질환 예방을 위해 유산소 운동과 식단 관리가 시급합니다."

```python
"""
파일명: health_service_lab.py
기능: 건강 데이터 입력 검증, DB 영구 보관, 동적 케어 가이드 종합 서비스
"""

from typing import Dict, Any
from health_calculator import calculate_health_metrics
from health_database import save_bmi_record, fetch_latest_bmi_record


def get_personalized_advice(bmi_status: str, bmr_kcal: float) -> str:
    """비만도 등급과 기초대사량을 기반으로 맞춤형 생활 수칙을 반환합니다."""
    base_advice = f"고객님의 하루 숨만 쉬어도 소모되는 기초대사량은 약 {bmr_kcal} kcal입니다.\n"
    
    if bmi_status == "저체중":
        return base_advice + "💡 신체 면역력 유지를 위해 균형 잡힌 단백질 위주의 식단과 가벼운 근력 운동을 권장합니다."
    elif bmi_status == "정상":
        return base_advice + "✨ 매우 이상적인 건강 상태입니다! 지금처럼 규칙적인 수면과 식습관을 유지해 주세요."
    elif bmi_status == "비만 전단계 (과체중)":
        return base_advice + "⚠️ 과체중 단계입니다. 매일 30분 빠르게 걷기와 식후 가벼운 산책을 시작해 보세요."
    else:
        return base_advice + "🚨 비만 단계입니다. 혈압 및 혈당 관리를 위해 기름진 식단을 줄이고 전문가 상담을 권장합니다."


def process_user_health_submission(
    user_email: str,
    raw_height: str,
    raw_weight: str,
    raw_age: str,
    gender: str = "남성",
    memo: str = ""
) -> Dict[str, Any]:
    """
    모바일 UI 폼에서 넘어온 원시 문자열 데이터를 안전하게 파싱하여 검증, 계산, 저장합니다.
    """
    # 1. 문자열을 숫자형으로 안전 변환
    try:
        height_val = float(str(raw_height).strip())
        weight_val = float(str(raw_weight).strip())
        age_val = int(str(raw_age).strip())
    except (ValueError, TypeError):
        return {
            "success": False,
            "message": "키, 몸무게, 나이는 정확한 숫자로 입력해야 합니다."
        }

    # 2. 신체 계산 및 유효 범위 체크
    try:
        metrics = calculate_health_metrics(height_val, weight_val, age_val, gender)
    except ValueError as ve:
        return {"success": False, "message": str(ve)}

    # 3. 데이터베이스 영구 저장
    db_result = save_bmi_record(
        user_email=user_email,
        height_cm=height_val,
        weight_kg=weight_val,
        age=age_val,
        gender=gender,
        memo=memo
    )

    # 4. 맞춤형 생활 조언 생성
    advice = get_personalized_advice(metrics["bmi_status"], metrics["bmr_kcal"])

    return {
        "success": True,
        "message": "신체 건강 지표가 정상적으로 분석 및 저장되었습니다.",
        "metrics": metrics,
        "advice": advice,
        "db_saved": db_result.get("success", False)
    }


# --- 종합 실습 테스트 실행 ---
if __name__ == "__main__":
    print("=" * 65)
    print("스마트119 건강 분석 대시보드 종합 서비스 실행")
    print("=" * 65)

    test_email = "student@smart119.school"
    result = process_user_health_submission(
        user_email=test_email,
        raw_height="178.4",
        raw_weight="82.5",
        raw_age="17",
        gender="남성",
        memo="체육 시간 전 측정"
    )

    print(f"처리 결과: {result['message']}")
    if result["success"]:
        m = result["metrics"]
        print(f"\n📊 [계측 결과]")
        print(f"- 체질량지수(BMI): {m['bmi']} ({m['bmi_status']})")
        print(f"- 기초대사량(BMR): {m['bmr_kcal']} kcal")
        print(f"\n💬 [맞춤형 건강 가이드]\n{result['advice']}")
        print(f"- 클라우드 DB 저장 상태: {'정상 완료' if result['db_saved'] else '실패'}")

```
